# --------------------------------------------------------------------------
# 1. HTTPS en www afdwingen
#    De oude site gebruikte sleutelbeheersystemen.nl zonder www, de nieuwe
#    site gebruikt www. Alles gaat naar één versie, zodat Google niet twee
#    afzonderlijke sites ziet.
# --------------------------------------------------------------------------
<IfModule mod_rewrite.c>
  RewriteEngine On

  RewriteCond %{HTTPS} off
  RewriteRule ^(.*)$ https://www.sleutelbeheersystemen.nl/$1 [R=301,L]

  RewriteCond %{HTTP_HOST} ^sleutelbeheersystemen\.nl$ [NC]
  RewriteRule ^(.*)$ https://www.sleutelbeheersystemen.nl/$1 [R=301,L]
</IfModule>

# --------------------------------------------------------------------------
# 3. Caching
#    PageSpeed klaagde over 10 minuten; dat was een beperking van GitHub
#    Pages. Hier zetten we het fatsoenlijk.
# --------------------------------------------------------------------------
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/webp             "access plus 1 year"
  ExpiresByType image/jpeg             "access plus 1 year"
  ExpiresByType image/png              "access plus 1 year"
  ExpiresByType image/svg+xml          "access plus 1 year"
  ExpiresByType image/x-icon           "access plus 1 year"
  ExpiresByType video/mp4              "access plus 1 year"
  ExpiresByType font/woff2             "access plus 1 year"
  ExpiresByType application/font-woff2 "access plus 1 year"
  ExpiresByType text/css               "access plus 1 month"
  ExpiresByType application/javascript "access plus 1 month"
  ExpiresByType text/javascript        "access plus 1 month"
  ExpiresByType application/pdf        "access plus 1 week"
  ExpiresByType text/html              "access plus 10 minutes"
</IfModule>

<IfModule mod_headers.c>
  <FilesMatch "\.(webp|jpg|jpeg|png|svg|ico|woff2|mp4)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
</IfModule>

# --------------------------------------------------------------------------
# 4. Compressie
# --------------------------------------------------------------------------
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/xml text/css
  AddOutputFilterByType DEFLATE application/javascript text/javascript
  AddOutputFilterByType DEFLATE application/json application/xml
  AddOutputFilterByType DEFLATE image/svg+xml
</IfModule>

# --------------------------------------------------------------------------
# 5. Beveiliging
# --------------------------------------------------------------------------
<IfModule mod_headers.c>
  Header set X-Content-Type-Options "nosniff"
  Header set X-Frame-Options "SAMEORIGIN"
  Header set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>

Options -Indexes

# de versleutelde stamboomgegevens niet laten indexeren
<FilesMatch "\.enc$">
  <IfModule mod_headers.c>
    Header set X-Robots-Tag "noindex, nofollow"
  </IfModule>
</FilesMatch>
